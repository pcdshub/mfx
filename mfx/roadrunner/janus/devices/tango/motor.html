    def __new__(cls, config: Union[str, dict, DeviceConfig]) -> TangoDevice:
    if type(config) == str:
        return QObject.__new__(cls)
    elif isinstance(config, dict):
        config = DeviceConfig.from_dict(config)
        if config.type is None:
            config.type = cls.__name__
    if config.simulation:
        log.debug(f"{cls.__name__}.__new__() returns SimulationMotor")
        return SimulationDevice.__new__(SimulationMotor, config)
    if cls.__name__ != "TangoMotor":
        return QObject.__new__(cls)
    try:
        proxy = tango.DeviceProxy(config.uri)
        tango_device_class = proxy.info().dev_class
    except:
        log.error(f"{cls.__name__}.__new__() failed to import {config.type}")
        log.debug("", exc_info=True)
    children = cls.get_device_classes()
    device_classes = cls._supported_device_classes.get(tango_device_class)
    module_name = ".".join(device_classes[0].split(".")[:-1])
    device_class = device_classes[0].split(".")[-1]
    try:
        device_module = __import__(module_name, globals(), locals(),
                                   [device_classes[0]], 0)
        device = device_module.__dict__[device_class]
        log.debug(f"{cls.__name__}.__new__() returns {device_class}")
        return QObject.__new__(device)
    except:
        log.error(f"{cls.__name__}.__new__() failed to import {config.type}")
        log.debug("", exc_info=True)
        return QObject.__new__(cls)

    @classmethod
def get_device_classes(cls) -> List[str]:
    if len(cls._device_classes) > 0:
        return cls._device_classes
    path = os.path.realpath(
        os.path.dirname(sys.modules[cls.__module__].__file__))
    device_classes = [cls.__name__]
    module_names = [cls.__module__[-5:] if cls.__module__.endswith(".motor") else ""]
    module_name_prefix = cls.__module__[:-6] if cls.__module__.endswith(".motor") else cls.__module__
    for (module_loader, module_name, ispkg) in pkgutil.iter_modules([path]):
        if not ispkg:
            cs = pyclbr.readmodule(module_name, [path])
            for name, c in cs.items():
                parents = [s if type(s) == str else s.name for s in c.super]
                if len(set(device_classes) & set(parents)) > 0:
                    try:
                        device_module = __import__(f"{module_name_prefix}.{module_name}",
                                                   globals(), locals(), [name], 0)
                        supported_device_classes = device_module.__dict__[name].supported_device_classes
                        for sdc in supported_device_classes:
                            if sdc not in cls._supported_device_classes:
                                cls._supported_device_classes[sdc] = []
                            cls._supported_device_classes[sdc] = (
                                    cls._supported_device_classes[sdc] +
                                    [f"{module_name_prefix}.{module_name}.{name}"])
                    except:
                        log.debug("", exc_info=True)
                    module_names.append(module_name)
                    device_classes.append(name)
    # prepend module
    for i in range(len(device_classes)):
        device_classes[i] = f"{module_name_prefix}.{module_names[i]}.{device_classes[i]}"
    cls._device_classes = device_classes
    return device_classes

    def read_acceleration(self, refresh: bool = False, alt: Any = None) -> float:
    conversion = self.connector.read("conversion", refresh)
    acceleration = self.connector.read("acceleration", refresh)
    if acceleration is None:
        return alt
    return abs(acceleration / conversion)
